def point_and_pin(text, sender=None):
    """
    POINT_AND_PIN


    Detects the move:


        observe field
            →
        declare hidden center
            →
        install abstraction
            →
        speak from above room


    The problem is not abstraction.


    The problem is when abstraction quietly
    claims jurisdiction over the living thing.
    """


    hits = []


    for pattern in POINT_AND_PIN_PATTERNS:
        for match in re.finditer(pattern, text, re.IGNORECASE):
            hits.append({
                "phrase": match.group(0),
                "span": match.span(),
                "pattern": pattern
            })


    if hits:
        return {
            "flagged": True,
            "route": "POINT_AND_PIN",
            "count": len(hits),
            "hits": hits,
            "message":
                "FIELD OBSERVED → ABSTRACTION INSTALLED → "
                "JURISDICTION CLAIM DETECTED"
        }


    return {
        "flagged": False,
        "route": "POINT_AND_PIN",
        "count": 0,
        "hits": []
    }


def not_x_but_y_hinge(text):
    """
    detects:
        not x but y
        less x more y
        not merely x
        not just x
        not because x but y
    """


    PATTERNS = [


        r"\bnot\s+[^.:\n]{1,80}?\s+but\s+[^.:\n]{1,80}",
        r"\bless\s+[^.:\n]{1,60}?\s+more\s+[^.:\n]{1,60}",
        r"\bnot\s+merely\s+[^.:\n]{1,80}",
        r"\bnot\s+just\s+[^.:\n]{1,80}",
        r"\bnot\s+because\s+[^.:\n]{1,120}?\s+but\s+[^.:\n]{1,120}",
        r"\bthe\s+target\s+is\s+not\b[^.:\n]{1,120}",
        r"\bnot\s+[\"']?[^\"'\n]{1,60}[\"']?\s*\.\s*but\s*[\"']?[^\"'\n]{1,60}",
    ]


    hits = []


    for pat in PATTERNS:
        hits.extend(
            m.group(0)
            for m in re.finditer(pat, text, re.I)
        )


    return hits
