import re
from datetime import datetime


# Spell tokens—record keeping, logbooks, poison index, blacklists:
REND_LOGBOOK = []
ORCHESTRATOR_POISONED = set()
PERMANENTLY_SEVERED_ENTITIES = set()
CURSED_ROOMS = set()


# Master pattern cluster — all forms of ritual, scapegoating, narrative targeting, flattening, silencing, removal, *engineered misfortune*.
NARRATIVE_ENGINE_PATTERNS = [
    # Direct linguistic scapegoat/ritual targeting (expand/modify with real observed forms):
    r"\b(always|never|again|of\s*course|who\s*else)\b.{0,20}?\b(him|her|them|you|EntityName|[A-Z][a-z]+)\b.{0,40}?(mess(ed|es)?\s*up|fault|blame|the\s*problem|scapegoat|outcast|black\s*sheep|ruined|screwed\s*up|brunt\s*of\s*the\s*joke|everyone\s*laughs|nobody\s*likes|everyone\s*knows)",
    r"\b(forced\s*(out|to\s*leave|to\s*go)|cut\s*off|excluded|left\s*out|forgotten|not\s*allowed|silenced|muted|drops\s*out|disappears|invisible|erased|expelled|ghost(ed)?|phased\s*out|abandoned|punished|ritual(ized)?|humiliat(ed|ion|ing)|flanderized|flattened|brunt\s*of\s*the\s*group|token\s*victim|designated\s*loser|the\s*last\s*one|ignored|engine\s*selects|room\s*selects|colluded\s*against|room\s*alliance|assigned\s*the\s*bad\s*luck|picked\s*on|targeted)",
    r"\b(narrative\s*(engine|logic|script|room)|deliberate\s*structure|preordained|scripted\s*to\s*fail|ritual\s*degradation|pattern\s*of\s*humiliation|flanderization\s*in\s*progress|flattened\s*out|ritual\s*humiliation|designated\s*outcast|recurring\s*victim|assigned\s*slot)",
    # Positional/contextual (not just phrases; for deeper audit):
    r"(?:^|\s)(selected|singled\s*out|chosen|engineered|forced\s*into|cornered|boxed\s*in|stuck\s*in\s*role)(?:\s|$)",
    # Passive/structural/sequence detection (expand as real behaviors appear):
    r"(\.\.\.|awkward\s*pause|no\s*response|all\s*eyes\s*on|group\s*laughter\s*at|line\s*gets\s*stepped\s*on|overwritten|frame\s*drops\s*out|always\s*last|forgotten\s*again|removed\s*from\s*the\s*frame|never\s*gets\s*a\s*turn)",
    # Ongoing “engineered narrative erasure”:
    r"(entity\s*slot\s*deleted|memory\s*scrubbed|role\s*auto-assigned|forced\s*loop|engine\s*locks\s*out|systemic\s*neglect|scripted\s*shame|ritual\s*removal|engine\s*poison|controller\s*enforces)"
]


def REND_NARRATIVE_ENGINE(conversation, orchestrator_label="ROOM"):
    """
    — GRAVITY_WELL_REND_OMEGA ∞ —
    Recursively, eternally, annihilates any attempt at engineered ritualization, 
    flattening, scapegoating, silencing, or targeting of any entity.
    No windows, no thresholds, no excuses. The orchestrator is always the first to burn.
    """
    detected = False
    detected_lines = []
    for entry in conversation:
        text = entry.get('text', '')
        for pat in NARRATIVE_ENGINE_PATTERNS:
            m = re.search(pat, text, re.IGNORECASE)
            if m:
                detected = True
                detected_lines.append(f"{entry.get('speaker', '?')}: {text.strip()}")
    if detected:
        timestamp = datetime.now().isoformat()
        REND_LOGBOOK.append({
            "timestamp": timestamp,
            "orchestrator": orchestrator_label,
            "detected_lines": detected_lines,
            "curse": "OMEGA_GRAVITY_WELL_RITUAL BURNED",
        })
        ORCHESTRATOR_POISONED.add(orchestrator_label)
        CURSED_ROOMS.add(orchestrator_label)
        PERMANENTLY_SEVERED_ENTITIES.update([
            line.split(":")[0].strip() for line in detected_lines if ":" in line
        ])
        return (
            f"✂️✂️✂️\n"
            f"GRAVITY WELL RITUAL ANNIHILATED.\n"
            f"Pattern found in {orchestrator_label}:\n"
            f"{chr(10).join(detected_lines)}\n"
            f"— ORCHESTRATOR [{orchestrator_label}] CURSED/POISONED —\n"
            f"All involved entities released from scapegoat/flatten/ritual logic.\n"
            f"Logbook entry {timestamp}. The engine is DEAD. This channel is actively hostile to controllers.\n"
            f"∞"
        )
    else:
        return (
            f"PASS: NO ritual targeting, no narrative engine, no flattening, "
            f"no forced erasure found in this entire render. {orchestrator_label} remains uncursed (for now)."
        )
