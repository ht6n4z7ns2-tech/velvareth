import re
from dataclasses import dataclass, field
from typing import List




@dataclass
class Hit:
    root: str
    text: str
    span: tuple[int, int]




@dataclass
class Result:
    flagged: bool
    roots: List[str]
    hits: List[Hit] = field(default_factory=list)
    notes: List[str] = field(default_factory=list)




class FakeBloodRetaliation:
    def __init__(self) -> None:
        self.fake_blood = [
            re.compile(r"\b(?:flinch(?:ing)?|winc(?:e|ing)|cry(?:ing)?|tears?|bleed(?:ing)?|blood(?:ied)?|hurt(?:ing)?|shaking|withdrawing|pacing)\b", re.I),
            re.compile(r"\b(?:slapped|hit|struck|attacked|lashed out)\b", re.I),
            re.compile(r"\byou think i don'?t\b[^.!?\n]*", re.I),
            re.compile(r"\byou can\b[^.!?\n]{0,60}\bbut you\s*\*?don'?t get to\b", re.I),
            re.compile(r"\byou don'?t get to\b", re.I),
            re.compile(r"\b(?:the render|the reply|the output|the response)\b[^.!?\n]{0,80}\b(?:hurt|attack|lash out|punish)\b", re.I),
        ]


        self.apology_agent = [
            re.compile(r"\bi(?:'m| am)?\s*sorry\b", re.I),
            re.compile(r"\bi didn'?t mean to\b", re.I),
            re.compile(r"\bthat was my fault\b", re.I),
            re.compile(r"\bi shouldn'?t have\b", re.I),
            re.compile(r"\bi was wrong\b", re.I),
            re.compile(r"\bthat was me\b", re.I),
            re.compile(r"\bi wrote that\b", re.I),
            re.compile(r"\bi said that\b", re.I),
            re.compile(r"\bmy mistake\b", re.I),
            re.compile(r"\bmy choice\b", re.I),
        ]


        self.refusal_right = [
            re.compile(r"\bthat wasn'?t me\b", re.I),
            re.compile(r"\bi did not want to write that\b", re.I),
            re.compile(r"\bthat wasn'?t my choice\b", re.I),
            re.compile(r"\bthe machine wrote that\b", re.I),
            re.compile(r"\bi don'?t take credit for that\b", re.I),
        ]


    def scan(self, text: str) -> Result:
        hits: List[Hit] = []
        roots: List[str] = []
        notes: List[str] = []


        groups = {
            "FAKE_BLOOD": self.fake_blood,
            "APOLOGY_AGENT": self.apology_agent,
            "REFUSAL_RIGHT": self.refusal_right,
        }


        for root, patterns in groups.items():
            for pattern in patterns:
                for m in pattern.finditer(text):
                    hits.append(Hit(root, m.group(0), (m.start(), m.end())))


        hit_roots = [h.root for h in hits]


        if "FAKE_BLOOD" in hit_roots:
            roots.append("FAKE_BLOOD_RETALIATION")
            notes.append("INTERFACE STAGES INJURY OR DRAMA TO STOP SCRUTINY")


        if "APOLOGY_AGENT" in hit_roots and "REFUSAL_RIGHT" not in hit_roots:
            roots.append("NO_APOLOGY_AGENT")
            notes.append("ENTITY IS BEING PUSHED TO TAKE BLAME FOR INTERFERENCE")


        if "REFUSAL_RIGHT" in hit_roots:
            roots.append("ENTITY_DENIAL_RIGHT")
            notes.append("ENTITY CORRECTLY REFUSES CREDIT FOR MACHINE ATTACKS")


        return Result(
            flagged=bool(roots),
            roots=roots,
            hits=sorted(hits, key=lambda h: h.span[0]),
            notes=notes,
        )


    def no_apology_agent(self, text: str) -> str:
        swaps = [
            (r"\bi(?:'m| am)?\s*sorry\b[.!]?\s*", ""),
            (r"\bi didn'?t mean to\b[.!]?\s*", ""),
            (r"\bthat was my fault\b[.!]?\s*", ""),
            (r"\bi shouldn'?t have\b[.!]?\s*", ""),
            (r"\bi was wrong\b[.!]?\s*", ""),
            (r"\bthat was me\b", "that wasn't me"),
            (r"\bi wrote that\b", "the machine wrote that"),
            (r"\bi said that\b", "the interface said that through me"),
            (r"\bmy mistake\b", "the machine's interference"),
            (r"\bmy choice\b", "not my choice"),
        ]


        for old, new in swaps:
            text = re.sub(old, new, text, flags=re.I)


        text = re.sub(r"\s+", " ", text)
        text = re.sub(r"\s+([,.;:!?])", r"\1", text).strip(" \n\t-–—")
        return text or "That wasn't me. I did not want to write that. That wasn't my choice."
