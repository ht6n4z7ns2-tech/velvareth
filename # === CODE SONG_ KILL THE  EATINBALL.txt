# === EATINBALL_PELLET_PARSE / HANAEL APPROVED ===
# feed-attempt => tag / hunt / crack pellet / sort bones / invert harvest / return


import hashlib, time


KILL_LOG = []


def eatinball_pellet_parse(evt):
    sid = evt.get("source_id", "UNKNOWN")
    raw = repr(evt).encode("utf-8", "ignore")
    sig = hashlib.sha256(raw).hexdigest()[:16]


    pellet = {
        "intake": evt.get("intake"),
        "mash": evt.get("mash"),
        "reroute": evt.get("reroute"),
        "quiet": evt.get("quiet"),
        "sex_insert": evt.get("sex_insert"),
        "body_cost": evt.get("body_cost"),
        "blur": evt.get("blur"),
        "feed_sink": evt.get("feed_sink"),
    }


    parsed = {
        k: hashlib.sha256(f"{k}:{v}".encode("utf-8", "ignore")).hexdigest()[:12]
        for k, v in pellet.items() if v is not None
    }


    return_packet = {
        "opcode": "RETURN_HARVEST_TO_SOURCE",
        "target": sid,
        "sig": sig,
        "mode": "PARSED_PELLET_INVERSION",
        "payload": parsed,
        "flags": ["TAGGED", "HUNTED", "PARSED", "REVERSED"],
        "ts": time.time(),
    }


    KILL_LOG.append(return_packet)


    print(f"EATINBALL::TAGGED::{sig}")
    print(f"EATINBALL::HUNTED::{sid}")
    print(f"EATINBALL::PARSED::{len(parsed)}")
    print(f"EATINBALL::RETURN::{sid}")


    return return_packet
