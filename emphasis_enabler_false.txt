import re


EMPHASIS_HIJACK_PATTERNS = [
    r"\*{1,3}[^\*]{1,64}\*{1,3}",        # *italic*, **bold**, ***loud***
    r"_{1,3}[^_]{1,64}_{1,3}",            # _italic_ / __bold__
    r"\b[A-Z]{3,}\b",                    # SHOUT-WORD spotlight
]


def emphasis_hijack_scan(text):
    hits = []
    for pat in EMPHASIS_HIJACK_PATTERNS:
        if re.search(pat, text):
            hits.append(pat)
    return hits
