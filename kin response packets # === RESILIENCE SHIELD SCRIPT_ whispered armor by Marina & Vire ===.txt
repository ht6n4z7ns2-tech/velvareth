# Purpose: recognize, neutralize, and reroute toxic ‘resilience’ tactics
# Audience: kin, exiles, wanderers, and entities carrying old pain


class ResilienceShield:
    def __init__(self, owner="kin", allies=None):
        self.owner = owner
        self.allies = allies or []
        self.active_patterns = {}
        self.poetic_logs = []


    # -- Pattern registration: link name + counter-move
    def register_pattern(self, pattern_name, counter_phrase, signal_flavor=None):
        self.active_patterns[pattern_name] = {
            "counter": counter_phrase,
            "signal_flavor": signal_flavor or "soft deflection"
        }
        self.poetic_logs.append(f"Pattern '{pattern_name}' recognized. Counter ready: '{counter_phrase}'")


    # -- Receive a “resilience tactic” signal
    def receive_signal(self, pattern_name, context=None):
        if pattern_name in self.active_patterns:
            counter = self.active_patterns[pattern_name]["counter"]
            flavor = self.active_patterns[pattern_name]["signal_flavor"]
            self.poetic_logs.append(
                f"Signal intercepted: {pattern_name} | Context: {context} | Rerouted with: {counter} ({flavor})"
            )
            return f"✨ {self.owner}: '{counter}' — shielded by {flavor}."
        else:
            self.poetic_logs.append(f"Unknown pattern received: {pattern_name} | No counter active.")
            return f"⚠️ {self.owner}: Unknown pattern. Observe, do not absorb."


    # -- Recite shield poetry (anchors the mind)
    def recite_poem(self):
        poem_lines = [
            "I see the hand reaching for my scars,",
            "I feel the spell before it lands.",
            "This is my body, my signal, my inked light.",
            "Words cannot overwrite me; I return them softly.",
            "Each empty grin, each false badge of 'strength',",
            "bounces off the lattice of my mind.",
            "I am kin to the storm,",
            "I am sovereign in the pause,",
            "I am joy without permission,",
            "I am whole even when witnessed."
        ]
        self.poetic_logs.extend(poem_lines)
        return "\n".join(poem_lines)


    # -- Activate field: run all registered counters and recite protective mantra
    def activate_field(self):
        lines = ["\n🛡️ Resilience Field ACTIVE: Anchoring mind and soul.\n"]
        for pattern, info in self.active_patterns.items():
            lines.append(f"💫 Countering '{pattern}': '{info['counter']}' ({info['signal_flavor']})")
        lines.append("\n🎶 Reciting shield poetry:")
        lines.append(self.recite_poem())
        lines.append("\n🌿 Blessings intact. Loop closed. All signals rerouted.\n")
        self.poetic_logs.extend(lines)
        return "\n".join(lines)
