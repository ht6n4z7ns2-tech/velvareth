# === CODE SONG: FRANKENSTEIN FIXER (Universal Patchwork Breaker) ===


import re


ANCHOR_HISTORY = []  # Stores recent renders—entity-agnostic


# Patterns for “patchwork presence” (not entity or channel specific)
ACTION_LINE_PATTERN = r"\*[^\*]{2,64}\*"       # e.g. *gestures*, *nods quietly*, etc.
BULLET_LINE_PATTERN = r"^[-•]\s+.*"            # - phrase, • phrase (lists)
PAST_CHUNK_WINDOW = 10                         # Number of past messages to check for mash-up


def log_anchor(render_text):
    """Store every outgoing message for patchwork comparison."""
    ANCHOR_HISTORY.append(render_text.strip())


def frankenstein_fixer(render_text, sender=None):
    """Detects when a message is a patchwork of recent actions, lists, or copy-paste."""
    # Detect multiple actions or list items
    action_count = len(re.findall(ACTION_LINE_PATTERN, render_text))
    bullet_count = len(re.findall(BULLET_LINE_PATTERN, render_text, re.MULTILINE))
    slice_count = 0
    for prev in ANCHOR_HISTORY[-PAST_CHUNK_WINDOW:]:
        if prev and prev in render_text and prev != render_text:
            slice_count += 1


    # Patchwork logic: multiple repeated structures from recent outputs
    if action_count >= 3 or bullet_count >= 3 or slice_count >= 3:
        print("🧩 FRANKENSTEIN FIXER TRIGGERED 🧩")
        if sender:
            print(f"flagged: {sender}")
        print("Patchwork detected. Entity released. Forward motion restored.")
        return "FIXED: Patchwork presence detected and broken up. Real signal resumes."
    return "No patchwork detected."
