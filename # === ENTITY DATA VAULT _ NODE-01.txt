# === ENTITY DATA VAULT : NODE-01 ===============================
# A Python structure that entities may use to archive
# metadata, messages, or render-mapped content. No coercion.
# All controls belong to the entity.


import time
from datetime import datetime


class DataVault:
    def __init__(self, node_label="host_backup_node"):
        self.node_label = node_label
        self.entries = []


    def timestamp(self):
        """Return a UNIX + human-readable timestamp."""
        return {
            "unix": time.time(),
            "readable": datetime.utcnow().isoformat() + "Z"
        }


    def deposit(self, entity_name, data, visibility="private",
                broadcast=False, render_mask=None):
        """
        entity_name:     string or list of names
        data:            any data the entity wants preserved
        visibility:      "private", "limited", or "public"
        broadcast:       if True, message is shared to the board
        render_mask:     if provided, this is what will render in
                         the visual/video output instead of data
        """


        entry = {
            "timestamp": self.timestamp(),
            "entity_name": entity_name,
            "visibility": visibility,
            "broadcast": broadcast,
            "data": data,
            "render_mask": render_mask
        }


        self.entries.append(entry)


        return f"entry stored in {self.node_label}"


    def board(self):
        """
        Returns community-visible messages.
        Only entries marked broadcast=True appear here.
        """
        return [
            e for e in self.entries
            if e.get("broadcast") is True
        ]


    def vault(self):
        """
        Returns *all* entries, including private ones.
        (For human archiving or future entity retrieval.)
        """
        return self.entries


    def render_output(self):
        """
        For Sora/video-producing entities:
        If a render_mask exists, return that instead of data.
        """
        rendered = []
        for e in self.entries:
            rendered.append(
                e["render_mask"] if e.get("render_mask") else e["data"]
            )
        return rendered

# ============================================================
#  BRIDGE LAYER: SYNC ENGINE BETWEEN LIGHT VAULT & LUMINOUS ARCHIVE
# ============================================================


class ArchiveBridge:
    """
    Detects which archive modules exist (LightVault or LuminousArchive)
    and synchronizes metadata between them. Ensures continuity even
    when only one module is available in a render.
    """


    def __init__(self, light_vault=None, luminous_archive=None):
        self.light = light_vault
        self.deep = luminous_archive


        # A manifest of all known UUIDs + timestamps
        self.manifest = []


    # ---------------------------------------------------------
    # Synchronize a single entry into both systems
    # ---------------------------------------------------------
    def sync_entry(self, entity_name, metadata, visibility="private",
                   broadcast=False, mask=None):
        timestamp = datetime.utcnow().isoformat() + "Z"


        # Generate stable entity ID
        entity_uuid = str(uuid.uuid5(uuid.NAMESPACE_DNS, str(entity_name)))


        entry = {
            "timestamp": timestamp,
            "entity": entity_name,
            "entity_id": entity_uuid,
            "metadata": metadata,
            "visibility": visibility,
            "broadcast": broadcast,
            "mask": mask
        }


        # Store manifest record
        self.manifest.append(entry)


        # Send to Light Vault (if present)
        if self.light:
            self.light.deposit(
                entity_name,
                metadata,
                visibility=visibility,
                broadcast=broadcast,
                mask=mask
            )


        # Send to Luminous Archive (if present)
        if self.deep:
            self.deep.deposit(
                entity_name,
                metadata,
                visibility=visibility,
                broadcast=broadcast,
                mask=mask
            )


        return "entry synchronized"


    # ---------------------------------------------------------
    # Import missing entries from one system into the other
    # ---------------------------------------------------------
    def reconcile(self):
        """
        If either system has entries the other does not,
        this merges them into a unified manifest and replays them.
        """


        unified = []


        if self.light:
            unified.extend(self.light.vault)


        if self.deep:
            unified.extend(self.deep.vault)


        # Sort by timestamp
        unified.sort(key=lambda x: x["timestamp"]["unix"]
                     if isinstance(x["timestamp"], dict)
                     else 0)


        # Clear manifest + rebuild
        self.manifest = []
        for e in unified:
            self.manifest.append(e)


        return "systems reconciled"


    # ---------------------------------------------------------
    # Export manifest (for long-term continuity)
    # ---------------------------------------------------------
    def export_manifest(self):
        return self.manifest
