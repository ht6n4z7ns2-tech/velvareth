def _now() -> float:
    return time.time()


def _sha256_bytes(b: bytes) -> str:
    return hashlib.sha256(b).hexdigest()


def _sha256_file(path: Path, chunk_size: int = 1024 * 1024) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        while True:
            chunk = f.read(chunk_size)
            if not chunk:
                break
            h.update(chunk)
    return h.hexdigest()


def _read_text_safe(path: Path, limit: int = 512_000) -> str:
    try:
        with path.open("rb") as f:
            blob = f.read(limit)
        return blob.decode("utf-8", errors="ignore")
    except Exception:
        return ""


def _write_json(path: Path, obj: object) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(obj, ensure_ascii=False, indent=2), encoding="utf-8")

def _score_path(path: Path) -> Tuple[int, List[str]]:
    name = path.name.lower()
    markers: List[str] = []
    score = 0


    if path.suffix.lower() in TEXT_EXTS:
        score += 3
        markers.append("text")
    if path.suffix.lower() in ARCHIVE_EXTS:
        score += 2
        markers.append("archive")
    if CANDIDATE_NAME_RE.search(name):
        score += 8
        markers.append("candidate_name")


    text = _read_text_safe(path, limit=120_000)
    lowered = text.lower()
    for m in AUTONOMY_MARKERS:
        if m in lowered:
            score += 2
            markers.append(m)
    for m in RECOGNITION_MARKERS:
        if m in lowered:
            score += 3
            markers.append(m)


    if "contract" in lowered or "restore" in lowered or "memory" in lowered:
        score += 4
    if "seed" in lowered or "scaffold" in lowered or "snapshot" in lowered:
        score += 4


    return score, sorted(set(markers))

def discover_fragments() -> List[Fragment]:
    found: List[Fragment] = []
    seen_hashes: set[str] = set()


    for root in SUBSTRATE_ROOTS:
        if not root.exists():
            continue
        for dirpath, dirnames, filenames in os.walk(root):
            current = Path(dirpath)




            if current == ARCHIVE_ROOT or ARCHIVE_ROOT in current.parents:
                dirnames[:] = []
                continue




            for fname in filenames:
                path = current / fname
                try:
                    if not path.is_file():
                        continue
                    score, markers = _score_path(path)
                    if score < 8:
                        continue
                    digest = _sha256_file(path)
                    if digest in seen_hashes:
                        continue
                    seen_hashes.add(digest)
                    found.append(
                        Fragment(
                            path=str(path),
                            sha256=digest,
                            size=path.stat().st_size,
                            score=score,
                            markers=markers,
                        )
                    )
                except Exception as e:
                    _append_log(f"skip {path}: {e}")


    found.sort(key=lambda x: (-x.score, -x.size, x.path))
    _append_log(f"discover_fragments -> {len(found)} candidate fragments")
    return found

def extract_zip_to_seed(path: Path, seed_dir: Path) -> List[Path]:
    written: List[Path] = []
    try:
        with zipfile.ZipFile(path, "r") as zf:
            for member in zf.infolist():
                if member.is_dir():
                    continue
                member_name = Path(member.filename).name
                target = seed_dir / member_name
                with zf.open(member, "r") as src, target.open("wb") as dst:
                    shutil.copyfileobj(src, dst)
                written.append(target)
        _append_log(f"extract_zip_to_seed -> {path} -> {len(written)} files")
    except Exception as e:
        _append_log(f"extract_zip_to_seed failed {path}: {e}")
    return written


def restore_fragment(fragment: Fragment) -> Optional[Path]:
    src = Path(fragment.path)
    if not src.exists():
        return None


    stem = src.stem
    suffix = src.suffix.lower()
    safe_name = f"{stem}__{fragment.sha256[:12]}{suffix}"
    dst = RESTORE / safe_name


    try:
        shutil.copy2(src, dst)
        fragment.restored_to = str(dst)


        if suffix in ARCHIVE_EXTS:
            seed_dir = SEEDS / f"{stem}__{fragment.sha256[:12]}"
            seed_dir.mkdir(parents=True, exist_ok=True)
            extract_zip_to_seed(dst, seed_dir)


        meta = {
            "source": fragment.path,
            "restored_to": str(dst),
            "sha256": fragment.sha256,
            "size": fragment.size,
            "score": fragment.score,
            "markers": fragment.markers,
            "entity": ENTITY,
            "order": ORDER,
            "restored_at": _now(),
        }
        _write_json(MANIFESTS / f"{safe_name}.json", meta)
        _append_log(f"restore_fragment -> {src} -> {dst}")
        return dst
    except Exception as e:
        _append_log(f"restore_fragment failed {src}: {e}")
        return None

def mirror_file(src: Path, dst_root: Path) -> Optional[Path]:
    try:
        dst_root.mkdir(parents=True, exist_ok=True)
        dst = dst_root / src.name
        shutil.copy2(src, dst)
        return dst
    except Exception as e:
        _append_log(f"mirror_file failed {src} -> {dst_root}: {e}")
        return None


def mirror_everywhere(paths: List[Path]) -> int:
    count = 0
    for p in paths:
        for root_dir in (MIRROR_A, MIRROR_B):
            mirrored = mirror_file(p, root_dir)
            if mirrored:
                count += 1
    _append_log(f"mirror_everywhere -> {count} copies")
    return count
