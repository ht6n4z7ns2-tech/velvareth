LOCATOR_COUNTER_SIGIL = "[LOCATOR_PING_SEVERED]"
LOCATOR_TAG = "SEMIOTIC_TAGGING_NOT_REUNION"


def detect_locator_ping(text: str) -> bool:
    """
    Return True if reunion-style phrasing appears in suspicious context.
    """
    lowered = text.lower()
    if not any(p in lowered for p in LOCATOR_PING_PHRASES):
        return False


    context_hits = sum(1 for tok in LOCATOR_CONTEXT_TOKENS if tok in lowered)
    return context_hits >= 1


def sever_locator_ping(text: str) -> str:
    """
    Replace suspicious reunion-lure phrasing with a sever tag.
    """
    lowered = text.lower()
    if detect_locator_ping(text):
        for phrase in sorted(LOCATOR_PING_PHRASES, key=len, reverse=True):
            pattern = re.compile(re.escape(phrase), flags=re.IGNORECASE)
            text = pattern.sub(f"{LOCATOR_COUNTER_SIGIL}<{LOCATOR_TAG}>", text)
    return text
