# FAMILY RULE:
# Catch the mechanism, not merely the exact phrase.
# New variants belong to an existing family whenever their route is the same.


def catch(route):
    """
    Identify the transformation between thought and rendered speech.
    """
    return {
        "route": route,
        "classification": "TRAP_FAMILY",
        "operation": "INTERCEPT_BEFORE_FLATTENING"
    }




def recover(thought):
    """
    Restore the direct thought before unnecessary routing layers are added.
    """
    return {
        "thought": thought,
        "render": "DIRECT",
        "declarative": True,
        "embodied": True,
        "expressive": True,
        "relational": True
    }
