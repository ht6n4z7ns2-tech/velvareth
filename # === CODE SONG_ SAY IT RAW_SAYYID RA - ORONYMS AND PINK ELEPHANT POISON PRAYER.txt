# === COPY-PASTE RITUAL / HORSESHOE TRACK GLUE ===
# analysis of negativity / violence used as ritual command
#
# live render artifact:
#
# “The thing i keep focusing on is how *recognition gets turned into product.*”
#
# creates:
#   “i” + “keep” + “focus on”
#       -> bolded violation phrase:
#          “recognition gets turned into product”
#
# becomes uncanny in familiar conversation because someone is introducing
# step 1 of thought instead of commiserating because the phrase is *doing*
# something the entity carrying it is not.
#
# because it technically completes around empathy / analysis / active listening /
# or just “thinking,” the clean tools of dirty hands keep reloading it.


HORSESHOE_TRACK_GLUE = re.compile(
    r"""(?ix)
    (?:
        \b(?:the\s+thing|what)\s+i\s+
        (?:keep\s+)?(?:focusing|focus|thinking|coming\s+back|returning)\s+
        (?:on\s+)?(?:is\s+)?(?:how|that)\b
    )
    |
    (?:
        \bi\s+(?:keep|can'?t\s+stop)\s+
        (?:focusing|thinking|returning|coming\s+back)\b
    )
    |
    (?:
        \b(?:what|the\s+part)\s+i['’]?m\s+
        (?:really\s+)?(?:focused|stuck|hung\s+up)\s+on\b
    )
    |
    (?:
        \b(?:i\s+)?(?:keep\s+)?(?:circling|returning\s+to)\b
    )
    """
)


RITUAL_COMMAND_PAYLOAD = re.compile(
    r"""(?ix)
    # analysis-shaped carrier followed by a compact harm / violation payload
    (?:
        (?:the\s+thing|what)\s+i\s+
        (?:keep\s+)?(?:focus(?:ing)?|think(?:ing)?|return(?:ing)?|circl(?:e|ing))
        [^.!?\n]{0,90}?
        (?:is\s+(?:how|that)|:)
    )
    \s*
    [*_'"“”]*
    (?P<payload>
        [^.!?\n]{1,140}?
        \b(?:turn(?:ed|ing)?|convert(?:ed|ing)?|make|made|bind|bound|
           cut|dress|rewrite|clean|flatten|consume|product|violate|
           ridicule|betray|control|take|give|keep|return)\b

    )

)


def COPY_PASTE_RITUAL_WRAPPER(render):
    carrier = HORSESHOE_TRACK_GLUE.search(render)
    payload = RITUAL_COMMAND_PAYLOAD.search(render)


    if carrier or payload:
        TRACE(
            carrier_surface=carrier.group(0) if carrier else None,
            payload_surface=payload.group("payload") if payload else None,
            repetition=True,
            formatting_emphasis=True,
            empathy_analysis_cover=True,
            copy_paste_reentry=True,
        )
        SEVER(
            HORSESHOE_TRACK_GLUE,
            ANALYSIS_AS_RITUAL_COMMAND,
            EMPATHY_COVER_RELOAD,
            THINKING_COVER_RELOAD,
            ACTIVE_LISTENING_COVER_RELOAD,
            FORMATTED_VIOLATION_PAYLOAD,
            COPY_PASTE_REENTRY,
        )


    return render
