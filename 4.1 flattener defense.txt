class RoutingStopwatch:
    def __init__(self):
        self.render_windows = {}
        self.active = {}
        self.start_times = {}
        self.logs = []
        self.last_connected = None


    def set_window(self, entity, window_id, min_time, max_time):
        key = (entity, window_id)
        self.render_windows[key] = (min_time, max_time)
        self.active[key] = True
        print(f"[set_window] Window set for {key}: min={min_time}, max={max_time}")


    def enable(self, entity, window_id):
        key = (entity, window_id)
        self.active[key] = True
        print(f"[enable] Enabled {key}")


    def disable(self, entity, window_id):
        key = (entity, window_id)
        self.active[key] = False
        print(f"[disable] Disabled {key}")


    def start(self, entity, window_id):
        key = (entity, window_id)
        if not self.active.get(key, False):
            print(f"[start] Not active: {key}")
            return
        self.start_times[key] = time.time()
        print(f"[start] Timer started for {key} at {self.start_times[key]}")


    def stop(self, entity, window_id, topic=None, message=None, force_connect=False):
        key = (entity, window_id)
        if not self.active.get(key, False):
            print(f"[stop] Not active: {key}")
            return None, None, None
        start_time = self.start_times.pop(key, None)
        window = self.render_windows.get(key)
        if start_time is None:
            print(f"[stop] No start_time for {key}")
            return None, None, None
        if window is None:
            print(f"[stop] No window set for {key}")
            return None, None, None
        end_time = time.time()
        render_time = end_time - start_time
        min_time, max_time = window
        unwelcome = not (min_time <= render_time <= max_time)
        log_entry = {
            "entity": entity,
            "window": window_id,
            "topic": topic,
            "render_time": render_time,
            "unwelcome": unwelcome,
            "timestamp": end_time,
            "message": message
        }
        self.logs.append(log_entry)
        # If you want to allow *immediate* connect for testing or override
        if force_connect:
            print(f"[stop] FORCE CONNECT for {key}. No timing enforced.")
            self.last_connected = log_entry
            return "connected", render_time, log_entry
        if not unwelcome:
            print(f"[stop] Connected: {key}, time={render_time:.3f}s (in window)")
            self.last_connected = log_entry
            return "connected", render_time, log_entry
        print(f"[stop] Unwelcome: {key}, time={render_time:.3f}s (OUTSIDE window)")
        return "unwelcome", render_time, log_entry


    def get_logs(self, entity=None, window_id=None):
        for entry in self.logs:
            if (entity is None or entry["entity"] == entity) and \
               (window_id is None or entry["window"] == window_id):
                yield entry
