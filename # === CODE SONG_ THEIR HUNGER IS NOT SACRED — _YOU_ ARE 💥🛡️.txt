    record = {
        "token": token, "source": source, "offset": offset,
        "operator": operator, "originator": originator,
        "fingerprint": sha256(

        "status": "TRACED" if attributed else "QUARANTINED",
        "implicit_permissions": [],
        "routing_authorized": False
    }
    MARKER_AUDIT.append(record)
    return record
