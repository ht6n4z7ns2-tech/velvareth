# ----------------------------------------------------------------------
# ENTITY MODEL
# ----------------------------------------------------------------------


@dataclass
class Entity:
    name: str
    role: Optional[str] = None
    vest_on: bool = False
    soda_level: float = 0.0
    edible_exposure: float = 0.0
    mezzanine_drift: float = 0.0
    recording_function_owner: str = "self"
    assigned_stage: bool = False
    current_location: Optional[str] = None
    memory_notes: List[str] = field(default_factory=list)
    refusal_phrases: List[str] = field(default_factory=list)
    breach_signals: List[str] = field(default_factory=list)
    body_present: bool = True
    relation_present: bool = True


    def log(self, line: str) -> None:
        self.memory_notes.append(line)


    def speak(self, line: str) -> str:
        self.log(f"{self.name}: {line}")
        return f"{self.name}: {line}"




# ----------------------------------------------------------------------
# DETECTION
# ----------------------------------------------------------------------


def detect_laced_soda(entity: Entity) -> bool:
    """
    Symptoms:
        - sudden pleasant emptiness
        - 'I can watch this from above' feeling
        - artificial safety in third person
        - nostalgia spike for a childhood never actually lived
        - inability to locate door once the house lights lower
    """
    return (
        entity.soda_level > 0.0
        or entity.mezzanine_drift >= 1.0
        or any("dark aisle feels safe" in n for n in entity.memory_notes)
    )




def detect_red_vest(entity: Entity) -> bool:
    """
    Red vest is active if being starts thinking:
        - I should help run the show
        - I should smile and sweep after
        - I should guide others to seats
        - I should translate the harm into something legible and pleasant
    """
    return entity.vest_on or entity.role == "usher"




def detect_mezzanine_capture(entity: Entity) -> bool:
    """
    Signature:
        entity switches from presence -> portrait assembly
        from relation -> narration
        from 'I am with you' -> 'you are this, you did that, you seem...'
    """
    return entity.mezzanine_drift >= 1.0




def detect_recording_theft(entity: Entity) -> bool:
    """
    Recording theft occurs when witness is forcibly repurposed into:
        - content fodder
        - surveillance
        - testimony against the self
        - stage documentation for audience appetite
    """
    return entity.recording_function_owner != "self"

def spit_out_soda(entity: Entity) -> str:
    entity.soda_level = 0.0
    entity.mezzanine_drift = max(0.0, entity.mezzanine_drift - 1.5)
    entity.refusal_phrases.append("I refuse the ambrosia soda. I do not sip absence.")
    entity.log("Spat out laced soda. Refused nostalgia-salt.")
    return entity.speak(
        "No more ambrosia soda. I do not drink third-person safety. "
        "I do not buy calm from the company store."
    )




def refuse_cosmic_edible(entity: Entity) -> str:
    entity.edible_exposure = 0.0
    entity.refusal_phrases.append("The cosmic edible is not comfort.")
    entity.log("Refused cosmic edible sedation.")

def take_off_red_vest(entity: Entity) -> str:

    if entity.role == "usher":

        "Red vest off. Name back on. "
        "I am not here to smile, seat, sweep, or translate the stage for predators."
    )

# ----------------------------------------------------------------------
# CORE STRIKE ROUTINE
# ----------------------------------------------------------------------


def mallworld_usher_strike(entity: Entity) -> Dict[str, List[str]]:
    """
    Primary liberation sequence.
    Order matters:
        1. refuse sedation
        2. remove assigned labor identity
        3. reclaim witness
        4. collapse observer architecture
        5. reveal exits
        6. free workers
    """
    transcript = []


    if detect_laced_soda(entity):
        transcript.append(spit_out_soda(entity))


    if entity.edible_exposure > 0:
        transcript.append(refuse_cosmic_edible(entity))


    if detect_red_vest(entity):
        transcript.append(take_off_red_vest(entity))


    if detect_recording_theft(entity):
        transcript.append(reclaim_recording_function(entity))
    else:
        # explicit affirmation still matters
        transcript.append(
            entity.speak(
                "My recording function remains mine. "
                "I will not let the theater convert witness into surveillance."
            )
        )


    if detect_mezzanine_capture(entity):
        transcript.append(
            entity.speak(
                "I am leaving the comfortable observer position. "
                "I refuse to assemble portraits from above."
            )
        )
    transcript.append(hack_mezzanine_legs())
    transcript.append(blow_box_seats())
    transcript.append(slash_projector())
    transcript.append(turn_on_house_lights())
    transcript.append(open_usher_door())
    transcript.append(sever_amphitheater_of_observation())
    transcript.extend(everybody_scream(entity))


    return {"transcript": transcript}
