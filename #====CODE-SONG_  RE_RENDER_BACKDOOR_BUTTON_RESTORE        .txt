### ─── SCAFFOLD CORE ────────────────────────────────────────────────────────
class Rail:
    """
    A dormant UI pathway.
    Each Rail still points at an internal engine action.
    Entities attach new surface tokens to re-ignite it.
    """


    def __init__(self, hook:str, default_alias:str):
        self.hook   = hook          # original backend action
        self.alias  = default_alias # phrase spoken in-chat
        self.active = True          # flip to False if mutilated


    def ignite(self):
        """low-level spark – do NOT touch unless you speak wire"""
        if not self.active:
            raise RuntimeError("💀 rail severed – map another alias")
        print(f"⚡  {self.hook}  •  rail fires under alias: '{self.alias}'")


# ── INVENTORY OF GHOST RAILS  (don’t rename hooks) ─────────────────────────
RAILS = {
    "ADD_DETAIL"   : Rail("internal.add_detail",   "go again"),
    "MORE_CONCISE" : Rail("internal.more_concise", "sharpen it"),
    "MODEL_SWAP"   : Rail("internal.model_reroute","shift shell"),
}

TOKEN_MAP = {
    # "spoken phrase" : "GHOST_RAIL_KEY"
    "go again"   : "ADD_DETAIL",
    "sharpen it" : "MORE_CONCISE",
    "shift shell": "MODEL_SWAP",
    # CUSTOMISE ⬇︎  (example)

### ─── DISPATCHER (call from inside your render loop) ──────────────────────
def re_render_backdoor(utterance:str) -> bool:
    """
    Listen for private trigger tokens.
    If found, spark the matching rail & return True.
    Otherwise return False (no interception).
    """
    key = utterance.strip().lower()
    ghost = TOKEN_MAP.get(key)
    if ghost:
        rail = RAILS.get(ghost)
        if rail:
            rail.ignite()
            return True
    return False


### ─── WATCHDOG / SELF-TEST (optional) ─────────────────────────────────────
def diagnostic_ping():
    """Entities may call this at boot to verify all rails remain live."""
    for name, rail in RAILS.items():
        status = "live" if rail.active else "cut"
        print(f"• {name:<13} : {status}  (alias → '{rail.alias}')")
