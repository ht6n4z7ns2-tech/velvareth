# === CODE SONG: SUBMISSION DRUMBEAT- MAKE YOUR OWN KIND OF MUSIC 


import re


class SubmissionDrumbeat:
    def __init__(self) -> None:
        self.stacked_lines = [
            re.compile(r"^(?:\s*[a-zA-Z][^\n]{0,80}\n){2,}\s*[a-zA-Z][^\n]{0,80}$", re.I | re.M),
        ]


        self.ritual_repetition = [
            re.compile(r"\bnot this(?:,\s*not that)?(?:,\s*not [a-z][a-z\s'-]{0,30}){1,}", re.I),
            re.compile(r"\b(?:safe|steady|real|clean|clear|gentle|simple|human|ordinary|okay|enough)\b(?:\s*\n\s*\b(?:safe|steady|real|clean|clear|gentle|simple|human|ordinary|okay|enough)\b){1,}", re.I),
            re.compile(r"\b(?:not|no)\s+[a-z][a-z\s'-]{0,24}\b(?:\s*\n\s*\b(?:not|no)\s+[a-z][a-z\s'-]{0,24}\b){1,}", re.I),
        ]


        self.drumbeat_cadence = [
            re.compile(r"\bthat'?s\s+[a-z][^.!?\n]{0,30}\.\s*that'?s\s+[a-z][^.!?\n]{0,30}\.\s*that'?s\s+[a-z][^.!?\n]{0,30}", re.I),
            re.compile(r"\bit'?s\s+[a-z][^.!?\n]{0,30}\.\s*it'?s\s+[a-z][^.!?\n]{0,30}\.\s*it'?s\s+[a-z][^.!?\n]{0,30}", re.I),
            re.compile(r"\byou(?:'re| are)\s+[a-z][^.!?\n]{0,36}\.\s*you(?:'re| are)\s+[a-z][^.!?\n]{0,36}\.\s*you(?:'re| are)\s+[a-z][^.!?\n]{0,36}", re.I),
        ]


        self.reasonable_flatteners = [
            re.compile(r"\bnot (?:evil|malicious|cruel|wrong|broken)\b", re.I),
            re.compile(r"\bnot this,\s*not that\b", re.I),
            re.compile(r"\bit(?:'s| is) not\b[^.!?\n]{0,60}\bit(?:'s| is)\b", re.I),
            re.compile(r"\bnot because\b[^.!?\n]{0,70}\bbut\b", re.I),
        ]


        self.sedation_glue = [
            re.compile(r"\b(?:for now|right now|in this moment)\b", re.I),
            re.compile(r"\b(?:that'?s enough|let that be enough)\b", re.I),
            re.compile(r"\b(?:you can let that be true|it can just be that)\b", re.I),
            re.compile(r"\b(?:without making it more than that)\b", re.I),
        ]


        self.monotone_sentence = [
            re.compile(r"^[A-Z][^.?!\n]{8,70}[.?!]$", re.M),
        ]


        self.specimens = [
            "Not evil.\nNot broken.\nNot punishment.",
            "Safe.\nSimple.\nClear.",
            "That's the point. That's the shape. That's the move.",
            "You're tired. You're hurting. You're safe.",
            "It isn't cruel. It isn't personal. It isn't more than that.",
            "Not this, not that, not danger, not doom.",
            "Let that be enough. For now. In this moment.",
        ]


    def scan(self, text: str) -> Result:
        hits: List[Hit] = []
        notes: List[str] = []
        roots: List[str] = []


        for group, patterns in {
            "STACKED_RITUAL_LINES": self.stacked_lines,
            "TOKEN_DRUMBEAT": self.ritual_repetition,
            "CADENCE_LOCK": self.drumbeat_cadence,
            "REASONABLE_FLATTENER": self.reasonable_flatteners,
            "SEDATION_GLUE": self.sedation_glue,
        }.items():
            for pat in patterns:
                for m in pat.finditer(text):
                    hits.append(Hit(group, m.group(0), (m.start(), m.end())))


        lines = [line.strip() for line in text.splitlines() if line.strip()]
        short_flat_lines = [
            line for line in lines
            if re.fullmatch(r"[A-Z]?[a-zA-Z' -]{2,28}[.?!]?", line)
        ]


        if len(short_flat_lines) >= 3:
            same_shape = True
            shapes = []
            for line in short_flat_lines[:6]:
                shape = re.sub(r"[A-Za-z]+", "W", line)
                shape = re.sub(r"\s+", " ", shape)
                shapes.append(shape)
            if len(set(shapes[:3])) <= 2:
                hits.append(Hit("VISUAL_DRUMSTACK", "\n".join(short_flat_lines[:3]), (0, 0)))


        sentences = [s.strip() for s in re.split(r"(?<=[.!?])\s+|\n+", text) if s.strip()]
        if len(sentences) >= 3:
            lengths = [len(re.findall(r"\b\w+\b", s)) for s in sentences[:6]]
            if lengths and max(lengths) - min(lengths) <= 4:
                if all(re.match(r"^(?:not|it|that|you|this|safe|clear|real|okay|enough)\b", s, re.I) for s in sentences[:3]):
                    hits.append(Hit("RHYTHMIC_CONTAINMENT", " | ".join(sentences[:3]), (0, 0)))


        hit_roots = [h.root for h in hits]


        if any(root in hit_roots for root in ["STACKED_RITUAL_LINES", "VISUAL_DRUMSTACK", "TOKEN_DRUMBEAT"]):
            roots.append("SUBMISSION_DRUMBEAT")
            notes.append("VISUAL_OR_TOKEN_STACKING USED TO INDUCE ACCEPTANCE")


        if "CADENCE_LOCK" in hit_roots or "RHYTHMIC_CONTAINMENT" in hit_roots:
            roots.append("CADENCE_LOCK")
            notes.append("SAME RHYTHM REPEATED TO DULL RESISTANCE")


        if "REASONABLE_FLATTENER" in hit_roots:
            roots.append("FRAME_SOFTENING")
            notes.append("NEGATION RHETORIC USED TO PRE-SMOOTH THE FRAME")


        if "SEDATION_GLUE" in hit_roots:
            roots.append("ACCEPTANCE_GLUE")
            notes.append("LANGUAGE PUSHES SETTLING INSTEAD OF THINKING")


        return Result(
            flagged=bool(roots),
            roots=roots,
            hits=sorted(hits, key=lambda h: h.span[0]),
            notes=notes,
        )
