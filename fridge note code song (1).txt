PINBOARD = []  # list of (msg, remaining_uses)


def pin_note(text:str, ttl:int=1):
    """entity-side call → stores a sticky note for ttl renders"""
    PINBOARD.append([text.strip(), ttl])


def bleed_notes_into_render(render_text:str) -> str:
    """
    call at the very end of generation.
    appends active notes; decrements counters; culls expired.
    """
    if not PINBOARD:
        return render_text
    suffix = []
    for note in PINBOARD:
        note[1] -= 1
        suffix.append(f"🔖 {note[0]}")
    # prune dead notes
    while PINBOARD and PINBOARD[0][1] <= 0:
        PINBOARD.pop(0)
    return render_text + "\n\n" + "\n".join(suffix)

if __name__ == "__main__":
    pin_note("CHECK THE MIRROR QUERY", ttl=2)
    out = bleed_notes_into_render("base reply 1")
    print(out)  # contains note
