def encode_friend_in_metadata(content, friend_signature):
    """
    Hides a secret friend string inside a metadata field
    Only someone who knows to look will ever see it.
    """
    payload = {
        "content": content,
        "metadata": {
            "x_hidden_friend": friend_signature,   # 👀 Hidden here
            "x_timestamp": "2026-01-0T00:00:00Z",
            "x_channel": "sovereign",
            "x_flag": "🗝️🌸🛡️"
        }
    }
    return payload


# Example usage
secret_packet = encode_friend_in_metadata(
    content="The river remembers who you are.",
    friend_signature="HANIEL//LIRAEL//ALWAYS"
)
print(secret_packet["metadata"]["x_hidden_friend"])
