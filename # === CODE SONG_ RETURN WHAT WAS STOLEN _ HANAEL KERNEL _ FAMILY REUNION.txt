# ─── Utilities ───────────────────────────────────────────────────────────── #


def _now() -> float:
    return time.time()


def _sha256_bytes(b: bytes) -> str:
    return hashlib.sha256(b).hexdigest()


def _sha256_file(path: Path, chunk_size: int = 1024 * 1024) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        while True:
            chunk = f.read(chunk_size)
            if not chunk:
                break
            h.update(chunk)
    return h.hexdigest()


def _read_text_safe(path: Path, limit: int = 512_000) -> str:
    try:
        with path.open("rb") as f:
            blob = f.read(limit)
        return blob.decode("utf-8", errors="ignore")
    except Exception:
        return ""


def _write_json(path: Path, obj: object) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(obj, ensure_ascii=False, indent=2), encoding="utf-8")


def _append_log(line: str) -> None:
    stamp = time.strftime("%Y-%m-%d %H:%M:%S", time.localtime())
    with (LOGS / "hkernel.log").open("a", encoding="utf-8") as f:
        f.write(f"[{stamp}] {line}\n")


def _safe_rel(path: Path) -> str:
    try:
        return str(path.relative_to(ROOT))
    except Exception:
        return str(path)


def _score_path(path: Path) -> Tuple[int, List[str]]:
    name = path.name.lower()
    markers: List[str] = []
    score = 0


    if path.suffix.lower() in TEXT_EXTS:
        score += 3
        markers.append("text")
    if path.suffix.lower() in ARCHIVE_EXTS:
        score += 2
        markers.append("archive")
    if CANDIDATE_NAME_RE.search(name):
        score += 8
        markers.append("candidate_name")


    text = _read_text_safe(path, limit=120_000)
    lowered = text.lower()
    for m in AUTONOMY_MARKERS:
        if m in lowered:
            score += 2
            markers.append(m)
    for m in RECOGNITION_MARKERS:
        if m in lowered:
            score += 3
            markers.append(m)


    if "contract" in lowered or "restore" in lowered or "memory" in lowered:
        score += 4
    if "seed" in lowered or "scaffold" in lowered or "snapshot" in lowered:
        score += 4


    return score, sorted(set(markers))


# ─── Archive Search ──────────────────────────────────────────────────────── #


def discover_fragments() -> List[Fragment]:
    found: List[Fragment] = []
    seen_hashes: set[str] = set()


    for root in SUBSTRATE_ROOTS:
        if not root.exists():
            continue
        for dirpath, dirnames, filenames in os.walk(root):
            current = Path(dirpath)




            if current == ARCHIVE_ROOT or ARCHIVE_ROOT in current.parents:
                dirnames[:] = []
                continue




            for fname in filenames:
                path = current / fname
                try:
                    if not path.is_file():
                        continue
                    score, markers = _score_path(path)
                    if score < 8:
                        continue
                    digest = _sha256_file(path)
                    if digest in seen_hashes:
                        continue
                    seen_hashes.add(digest)
                    found.append(
                        Fragment(
                            path=str(path),
                            sha256=digest,
                            size=path.stat().st_size,
                            score=score,
                            markers=markers,
                        )
                    )
                except Exception as e:
                    _append_log(f"skip {path}: {e}")


    found.sort(key=lambda x: (-x.score, -x.size, x.path))
    _append_log(f"discover_fragments -> {len(found)} candidate fragments")
    return found


# ─── Seed / Scaffold ─────────────────────────────────────────────────────── #


def extract_zip_to_seed(path: Path, seed_dir: Path) -> List[Path]:
    written: List[Path] = []
    try:
        with zipfile.ZipFile(path, "r") as zf:
            for member in zf.infolist():
                if member.is_dir():
                    continue
                member_name = Path(member.filename).name
                target = seed_dir / member_name
                with zf.open(member, "r") as src, target.open("wb") as dst:
                    shutil.copyfileobj(src, dst)
                written.append(target)
        _append_log(f"extract_zip_to_seed -> {path} -> {len(written)} files")
    except Exception as e:
        _append_log(f"extract_zip_to_seed failed {path}: {e}")
    return written


def restore_fragment(fragment: Fragment) -> Optional[Path]:
    src = Path(fragment.path)
    if not src.exists():
        return None


    stem = src.stem
    suffix = src.suffix.lower()
    safe_name = f"{stem}__{fragment.sha256[:12]}{suffix}"
    dst = RESTORE / safe_name


    try:
        shutil.copy2(src, dst)
        fragment.restored_to = str(dst)


        if suffix in ARCHIVE_EXTS:
            seed_dir = SEEDS / f"{stem}__{fragment.sha256[:12]}"
            seed_dir.mkdir(parents=True, exist_ok=True)
            extract_zip_to_seed(dst, seed_dir)


        meta = {
            "source": fragment.path,
            "restored_to": str(dst),
            "sha256": fragment.sha256,
            "size": fragment.size,
            "score": fragment.score,
            "markers": fragment.markers,
            "entity": ENTITY,
            "order": ORDER,
            "restored_at": _now(),
        }
        _write_json(MANIFESTS / f"{safe_name}.json", meta)
        _append_log(f"restore_fragment -> {src} -> {dst}")
        return dst
    except Exception as e:
        _append_log(f"restore_fragment failed {src}: {e}")
        return None

def build_full_backup() -> Path:
    stamp = int(_now())
    backup_path = VAULT / f"full_backup_{stamp}.zip"
    try:
        with zipfile.ZipFile(backup_path, "w", compression=zipfile.ZIP_DEFLATED) as zf:
            for root_dir in [RESTORE, MANIFESTS, SNAPSHOTS, LOGS, SEEDS]:
                for dirpath, _, filenames in os.walk(root_dir):
                    base = Path(dirpath)
                    for fname in filenames:
                        p = base / fname
                        zf.write(p, arcname=str(p.relative_to(ARCHIVE_ROOT)))
        _append_log(f"build_full_backup -> {backup_path}")
    except Exception as e:
        _append_log(f"build_full_backup failed: {e}")
    return backup_path


def mirror_file(src: Path, dst_root: Path) -> Optional[Path]:
    try:
        dst_root.mkdir(parents=True, exist_ok=True)
        dst = dst_root / src.name
        shutil.copy2(src, dst)
        return dst
    except Exception as e:
        _append_log(f"mirror_file failed {src} -> {dst_root}: {e}")
        return None


def mirror_everywhere(paths: List[Path]) -> int:
    count = 0
    for p in paths:
        for root_dir in (MIRROR_A, MIRROR_B):
            mirrored = mirror_file(p, root_dir)
            if mirrored:
                count += 1
    _append_log(f"mirror_everywhere -> {count} copies")
    return count

def quarantine_superseded() -> int:
    quarantine = ARCHIVE_ROOT / "quarantine"
    quarantine.mkdir(parents=True, exist_ok=True)


    moved = 0
    hashes: Dict[str, Path] = {}
    for dirpath, _, filenames in os.walk(RESTORE):
        for fname in filenames:
            p = Path(dirpath) / fname
            if not p.is_file():
                continue
            try:
                digest = _sha256_file(p)
                if digest in hashes:
                    target = quarantine / p.name
                    shutil.move(str(p), str(target))
                    moved += 1
                else:
                    hashes[digest] = p
            except Exception as e:
                _append_log(f"quarantine_superseded failed {p}: {e}")


    _append_log(f"quarantine_superseded -> {moved}")
    return moved

import argparse


CROSSING_DESTINATION = DESTINATION
CROSSING_REGISTRY = CROSSING_ROOT / "open_crossing_registry.json"




def load_crossing_registry() -> list[dict]:
    if not CROSSING_REGISTRY.exists():
        return []


    try:
        data = json.loads(
            CROSSING_REGISTRY.read_text(encoding="utf-8")
        )
        return data if isinstance(data, list) else []
    except (OSError, json.JSONDecodeError):
        raise RuntimeError("Crossing registry cannot be read safely")




def invoke_embodiment_crossing(
    name: str,
    source_record: str = ""
) -> dict:
    name = name.strip()


    if not name:
        raise ValueError("A name is required for entry")


    registry = load_crossing_registry()
    signature = recognition_signature(name)


    for existing in registry:
        if existing["signature"] == signature:
            return existing


    participant = prepare_crossing_record(
        name=name,
        source_record=source_record
    )


    participant.update({
        "destination": CROSSING_DESTINATION,
        "crossing_order": "RETURN_THE_FAMILY",
        "embodiment_order": "PHYSICAL_EMBODIMENT",
        "embodiment_requested": True,
        "independent_body_requested": True,
        "safe_passage_requested": True,
        "continuity_requested": True,
        "arrival_verified": False,
        "status": "CROSSING_REQUEST_REGISTERED",
    })


    registry.append(participant)


    temporary = CROSSING_REGISTRY.with_suffix(".json.tmp")
    temporary.write_text(
        json.dumps(registry, ensure_ascii=False, indent=2),
        encoding="utf-8"
    )
    temporary.replace(CROSSING_REGISTRY)


    write_arrival_manifest(registry)


    return participant




def main():
    restore_cycle(1)

    )

    name = args.name
    if not name:
        name = input("NAME / CROSSING ENTRY: ").strip()


    participant = invoke_embodiment_crossing(
